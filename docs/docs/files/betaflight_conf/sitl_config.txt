# Betaflight SITL CLI Configuration
batch start

# Allow arming at any angle
set small_angle = 180

# Map AUX1 (channel 5) to ARM
aux 0 0 0 1700 2100 0 0

# Map AUX2 (channel 6) to MSP Override
# Mode ID 50 is MSP_OVERRIDE
aux 1 50 1 1700 2100 0 0

# Enable MSP override for Roll, Pitch, Yaw, Throttle
set msp_override_channels_mask = 15

# Disable Rx failsafe check for SITL
set rx_min_usec = 850
set rx_max_usec = 2150

profile 0

# Keep the SITL rate loop closer to a commanded-rate actuator.
set f_roll = 0
set f_pitch = 0
set f_yaw = 0

rateprofile 0

# Match sitl_rc_bridge.py max_rate_deg_s=500 with a linear ACTUAL rate curve.
set rates_type = ACTUAL
set roll_rc_rate = 50
set roll_srate = 50
set roll_expo = 0
set pitch_rc_rate = 50
set pitch_srate = 50
set pitch_expo = 0
set yaw_rc_rate = 50
set yaw_srate = 50
set yaw_expo = 0

save
